        // تابع تولید کارت

        function renderCards(data, containerId) {
            const container = document.getElementById(containerId);
            if (!container) return;
            let html = '';
            data.forEach(item => {
                const genreTags = item.genre.map(g => `<span class="genre-tag">${g}</span>`).join('');
                html += `
                    <div class="card">
                        <div class="card-header">
                            <div class="card-title">
                                ${item.titleFa}
                                <span class="card-title-en">${item.titleEn}</span>
                            </div>
                            <span class="card-year">${item.year}</span>
                        </div>
                        <div class="card-genre">${genreTags}</div>
                        <div class="card-synopsis">${item.synopsis}</div>
                        <div class="card-poster">${item.poster}</div>
                    </div>
                `;
            });
            container.innerHTML = html;
        }
